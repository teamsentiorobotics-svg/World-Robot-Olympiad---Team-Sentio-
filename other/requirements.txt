# ----------------------------------------------------------------------
# Raspberry Pi system dependencies
# ----------------------------------------------------------------------
#
# Raspberry Pi-specific libraries are installed through Raspberry Pi OS
# rather than through pip.
#
# Install the required system packages using:
#
#   sudo apt update
#   sudo apt install -y python3-picamera2 python3-opencv \
#       python3-numpy i2c-tools
#
# Raspberry Pi 5 GPIO compatibility:
#
#   sudo apt remove -y python3-rpi.gpio
#   sudo apt install -y python3-rpi-lgpio
#
# rpi-lgpio provides compatibility with the GPIO interface used by the
# competition software:
#
#   import RPi.GPIO as GPIO
#
# Picamera2 is provided through Raspberry Pi OS and is imported as:
#
#   from picamera2 import Picamera2
#
# Standard-library modules such as time are included with Python and
# therefore do not require separate installation.
